msb-operator {{ .Chart.AppVersion }} installed into namespace {{ .Release.Namespace }}.

Components:
  - msb-controller  Deployment ({{ .Values.controller.replicas }} replicas, leader-elected)
  - msb-daemon      DaemonSet  (KVM device plugin, one per node)
{{- if .Values.installCRD }}
  - Sandbox CRD     sandboxes.sandbox.microsandbox.dev
{{- end }}

Check rollout:
  kubectl -n {{ .Release.Namespace }} rollout status deploy/msb-controller
  kubectl -n {{ .Release.Namespace }} get ds msb-daemon
  kubectl get node -o jsonpath='{.items[*].status.allocatable.devices\.microsandbox\.dev/kvm}'

Create a sandbox:
  cat <<EOF | kubectl apply -f -
  apiVersion: sandbox.microsandbox.dev/v1alpha1
  kind: Sandbox
  metadata:
    name: hello
  spec:
    image: alpine:3.20
    cpus: 1
    memory: 512
    cmd: ["echo", "hello from a microVM"]
    runPolicy: Once
  EOF

  kubectl get sandbox hello -w
