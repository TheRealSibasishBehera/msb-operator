  kubectl -n {{ .Release.Namespace }} get ds msb-daemon
  kubectl get node -o jsonpath='{.items[*].status.allocatable.devices\.microsandbox\.io/kvm}'

Create a sandbox:
  cat <<EOF | kubectl apply -f -
  apiVersion: sandbox.microsandbox.io/v1alpha1
  kind: Sandbox
  metadata:
    name: hello
  spec:
    image: alpine:3.20
    cpus: 1
    memory: 512
    cmd: ["echo", "hello from a microVM"]
    runPolicy: Once
  EOF

  kubectl get sandbox hello -w
